cmake_minimum_required(VERSION 3.24)
project(Trigger VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_compile_options(/W4 /utf-8 /permissive- /EHsc)
add_library(trigger_core src/core.cc)
target_include_directories(trigger_core PUBLIC src)
target_include_directories(trigger_core SYSTEM PUBLIC vendor)
add_executable(Trigger.CoreTests tests/core_tests.cc)
target_link_libraries(Trigger.CoreTests PRIVATE trigger_core)
enable_testing()
add_test(NAME core COMMAND Trigger.CoreTests)
add_library(trigger_desktops src/desktops.cc)
target_link_libraries(trigger_desktops PUBLIC trigger_core ole32 advapi32 user32)
add_executable(Trigger.DesktopTests tests/desktop_tests.cc)
target_link_libraries(Trigger.DesktopTests PRIVATE trigger_desktops)
add_test(NAME desktops COMMAND Trigger.DesktopTests)
add_library(trigger_platform src/platform.cc)
target_link_libraries(trigger_platform PUBLIC trigger_desktops shell32 shcore dwmapi crypt32 winhttp comdlg32 uuid)
add_executable(Trigger.WindowProbe WIN32 tests/window_probe.cc)
target_link_libraries(Trigger.WindowProbe PRIVATE shell32)
add_executable(Trigger.PlatformTests tests/platform_tests.cc)
target_link_libraries(Trigger.PlatformTests PRIVATE trigger_platform)
add_dependencies(Trigger.PlatformTests Trigger.WindowProbe)
add_test(NAME platform COMMAND Trigger.PlatformTests)
add_executable(Trigger WIN32 src/main.cc src/app.manifest)
target_include_directories(Trigger SYSTEM PRIVATE vendor/webview2/build/native/include)
target_link_libraries(Trigger PRIVATE trigger_platform "${CMAKE_CURRENT_SOURCE_DIR}/vendor/webview2/build/native/x64/WebView2LoaderStatic.lib" version)
add_custom_command(TARGET Trigger POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
  "${CMAKE_CURRENT_SOURCE_DIR}/web" "$<TARGET_FILE_DIR:Trigger>/web")
add_executable(Trigger.TestControl tests/test_control.cc)
target_link_libraries(Trigger.TestControl PRIVATE trigger_platform windowscodecs gdi32)
